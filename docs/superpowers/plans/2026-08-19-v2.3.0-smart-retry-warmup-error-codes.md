```python
"""IMP-05: 重试策略 - 自适应错误分类、指数退避 + jitter + Retry-After 支持。

为 worker 提供统一的错误分类和自适应退避计算。
支持基于错误类型配置不同退避策略和最大重试次数。
"""
import random
import re
import time
from typing import ClassVar


class AdaptiveRetryStrategy:
    """自适应重试策略：根据错误类型和上游反馈动态调整退避。

    ERROR_TYPES 定义每种错误类型的退避基值、最大重试次数、jitter 系数。
    delay() 执行指数退避：delay = min(backoff_base * 2^(attempt-1), max_delay)
    """

    ERROR_TYPES: ClassVar[dict[str, dict]] = {
        "rate_limited": {"backoff_base": 30, "max_retries": 5, "jitter": 0.5},
        "token_rejected": {"backoff_base": 5, "max_retries": 3, "jitter": 0.2},
        "timeout": {"backoff_base": 10, "max_retries": 3, "jitter": 0.3},
        "server_error": {"backoff_base": 15, "max_retries": 4, "jitter": 0.4},
        "network_error": {"backoff_base": 5, "max_retries": 2, "jitter": 0.5},
    }

    MAX_DELAY = 300.0  # 所有退避上限 300 秒

    # 错误类型判定标记（按优先级）
    _RATE_LIMITED_MARKERS: ClassVar[tuple[str, ...]] = (
        "429", "rate_limit", "rate limit", "too many requests",
    )
    _TOKEN_REJECTED_MARKERS: ClassVar[tuple[str, ...]] = (
        "token_rejected", "human verification failed",
    )
    _TIMEOUT_MARKERS: ClassVar[tuple[str, ...]] = (
        "timeout", "timed out",
    )
    _SERVER_ERROR_MARKERS: ClassVar[tuple[str, ...]] = (
        "5xx", "503", "502", "504", "500",
    )
    _NETWORK_ERROR_MARKERS: ClassVar[tuple[str, ...]] = (
        "connectionerror", "connection refused", "connection reset",
    )
    _PERMANENT_MARKERS: ClassVar[tuple[str, ...]] = (
        "422", "404",
    )

    # 兼容旧版 RetryPolicy 接口
    TRANSIENT_TYPES = frozenset(("rate_limited", "token_rejected", "timeout", "server_error", "network_error"))

    @classmethod
    def classify(cls, error: object) -> str:
        """基于错误消息和类型分类错误。

        返回值为 PERMANENT_MARKERS: permanent, 否则为 ERROR_TYPES 中的 key 之一。
        规则（按优先级，首个匹配为准）：
        - rate_limited: 429, rate_limit, too many requests
        - token_rejected: token_rejected, human verification failed
        - timeout: timeout, timed out
        - server_error: 5xx, 503, 502, 504, 500
        - network_error: ConnectionError, connection refused, connection reset
        - permanent: 422, 404
        - 默认: timeout（保守重试）
        """
        msg = str(error).lower()

        # 检查 httpx 响应状态码
        try:
            status = getattr(error, "response", None)
            if status is not None:
                status_code = getattr(status, "status_code", None) or getattr(status, "status", None)
                if status_code is not None:
                    if status_code == 429 or ("rate_limit" in msg or "rate limit" in msg):
                        return "rate_limited"
                    if status_code in (422, 404):
                        return "permanent"
                    if 500 <= status_code < 600:
                        return "server_error"
                    if status_code == 400 and ("rate_limit" in msg or "rate limit" in msg):
                        return "rate_limited"
        except (AttributeError, TypeError):
            pass

        # 字符串匹配
        for marker in cls._RATE_LIMITED_MARKERS:
            if marker in msg:
                return "rate_limited"
        for marker in cls._TOKEN_REJECTED_MARKERS:
            if marker in msg:
                return "token_rejected"
        for marker in cls._TIMEOUT_MARKERS:
            if marker in msg:
                return "timeout"
        if isinstance(error, (TimeoutError, ConnectionError)):
            return "network_error"
        for marker in cls._SERVER_ERROR_MARKERS:
            if marker in msg:
                return "server_error"
        if isinstance(error, ConnectionError):
            return "network_error"
        for marker in cls._NETWORK_ERROR_MARKERS:
            if marker in msg:
                return "network_error"
        for marker in cls._PERMANENT_MARKERS:
            if marker in msg:
                return "permanent"

        # 默认保守：timeout（可重试）
        return "timeout"

    @classmethod
    def should_retry(cls, attempt: int, max_retries: int, error: object) -> bool:
        """判断是否应重试。attempt 从 1 开始计数。

        兼容旧版接口：permanent 错误即使 attempt < max_retries 也不重试。
        """
        if attempt >= max_retries:
            return False
        return cls.classify(error) != "permanent"

    @classmethod
    def delay(cls, attempt: int, error_type: str) -> float:
        """指数退避 + jitter 延迟计算。

        attempt 从 1 开始计数（第 1 次重试）。
        delay = min(backoff_base * 2^(attempt-1), MAX_DELAY)
        最终结果 = delay + random(0, delay * jitter)
        """
        config = cls.ERROR_TYPES.get(error_type, cls.ERROR_TYPES["timeout"])
        base = min(config["backoff_base"] * (2 ** (attempt - 1)), cls.MAX_DELAY)
        return base + random.uniform(0, base * config["jitter"])

    @classmethod
    def delay_from_retry_after(cls, header_value: str | None) -> float | None:
        """解析 Retry-After 头，返回退避秒数。若值不合规则返回 None。"""
        if not header_value:
            return None
        # 尝试秒数
        try:
            return float(header_value.strip())
        except ValueError:
            pass
        # 尝试 HTTP-date（RFC 7231）
        try:
            from email.utils import parsedate_to_datetime
            dt = parsedate_to_datetime(header_value.strip())
            return max(0, (dt.timestamp() - time.time()))
        except Exception:
            return None

    # ── 兼容旧版 RetryPolicy 接口 ────────────────────

    @classmethod
    def classify_error(cls, error: object) -> str:
        """兼容旧版接口：返回 'transient' 或 'permanent'。"""
        return "permanent" if cls.classify(error) == "permanent" else "transient"

    @classmethod
    def backoff_delay(cls, attempt: int, base_delay: float) -> float:
        """兼容旧版接口：随机退避 [0, base_delay * 2^(attempt-1))。"""
        max_delay = base_delay * (2 ** (attempt - 1))
        return random.uniform(0, max_delay)


# 别名，兼容旧导入

"""F-05: 缓存预热 — 启动时从 DB 加载常见查询到缓存，减少重启后 DB 读压。"""
import logging

log = logging.getLogger("cache_warmup")


async def warmup_cache(gallery_cache, db) -> dict[str, int]:
    """启动后预热缓存：按优先级从 DB 加载常见查询，异步非阻塞。

    预热不影响启动速度（lifespan 中异步执行，不阻塞 yield）。
    返回 {key: count} 预热结果。
    """
    result: dict[str, int] = {}
    try:
        # 1. 统计概览（立即预热）
        overview = db.stats_overview()
        await gallery_cache.set("stats:overview", overview)
        result["stats:overview"] = 1
        log.info("缓存预热: stats:overview")
    except Exception as e:
        log.warning("缓存预热 stats:overview 失败: %s", e)

    try:
        # 2. 近日统计（14 天 + 12 月）
        daily = db.stats_daily(14)
        await gallery_cache.set("stats:daily:14", daily)
        result["stats:daily:14"] = 1
        monthly = db.stats_monthly(12)
        await gallery_cache.set("stats:monthly:12", monthly)
        result["stats:monthly:12"] = 1
        log.info("缓存预热: stats 条目")
    except Exception as e:
        log.warning("缓存预热 stats 失败: %s", e)

    try:
        # 3. 画廊（常见 limit 值）
        from . import config
        for limit in [10, 20, 50]:
            items = db.recent_images(limit)
            cache_key = f"gallery:{limit}"
            await gallery_cache.set(cache_key, {"items": items, "count": len(items)})
            result[cache_key] = 1
        log.info("缓存预热: gallery 条目")
    except Exception as e:
        log.warning("缓存预热 gallery 失败: %s", e)

    log.info("缓存预热完成: %d 个条目", len(result))
    return result
```

    """获取多语言错误消息，支持动态参数插值。

    按 lang → zh → 硬编码回退 顺序查找。
    """
    messages = ERROR_MESSAGES.get(lang, ERROR_MESSAGES.get("zh", {}))
    template = messages.get(code, "")
    if template:
        try:
            return template.format(**kwargs)
        except KeyError:
            return template
    return code  # 回退：返回错误码本身


class AppError(Exception):
    """统一的应用错误。"""

    def __init__(
        self,
        code: str,
        message: str,
        status_code: int = 400,
        details: dict | None = None,
    ):
        # 兼容旧版错误码（如 "QUEUE_FULL" → "SYS.002"）
        self.code = _LEGACY_CODE_MAP.get(code, code)
        self.message = message
        self.status_code = status_code
        self.details = details or {}
        super().__init__(self.message)


def error_response(
    code: str,
    message: str,
    status_code: int = 400,
    details: dict | None = None,
) -> JSONResponse:
    """统一错误响应格式。"""
    # 兼容旧版错误码
    mapped_code = _LEGACY_CODE_MAP.get(code, code)
    return JSONResponse(
        status_code=status_code,
        content={
            "error": {
                "code": mapped_code,
                "message": message,
                "details": details or {},
            }
        },
    )
